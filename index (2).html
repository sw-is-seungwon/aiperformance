① 데이터 불러오기 및 X, y 설정

# 데이터를 불러와 봅시다.
import pandas as pd
df = pd.read_csv("펭귄_데이터(처리전).csv")

df

# 펭귄의 키와 날개길이, 몸무게를 활용하여 부리길이를 예측하도록 변수를 설정해 봅시다.

# 독립 변수
X = df [['키_m', '날개길이_cm', '몸무게_kg']]

# 종속 변수
y = df['부리길이_cm']

② 정규화

# 정규화를 진행해 봅시다.

# 정규화에 필요한 기능 불러오기
from sklearn.preprocessing import MinMaxScaler

# X 데이터를 배열 형태로 변환
data_array = X.values

# 최소-최대 정규화 방법을 사용할 틀 제작
scaler = MinMaxScaler()

# X 데이터를 0~1 사이의 값으로 정규화한 값을 변수에 저장
normalized_data = scaler.fit_transform(data_array)

# 정규화한 데이터를 다시 데이터 프레임으로 변환(기존 X의 열 이름을 그대로 사용)
X = pd.DataFrame(normalized_data, columns = X.columns)

X

③ 훈련·테스트 데이터 분리

# 테스트 나누기에 필요한 기능 불러오기
from sklearn.model_selection import train_test_split

# 아래에 코드를 채워 봅시다.
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.3, # 60개의 데이터가 있을 때 테스트 데이터 18개
    random_state=42
)

len(X_test)

④ 모델 생성 및 학습

# 선형회귀 모델을 생성하고, 학습을 진행하는 코드를 작성해 봅시다.

# 선형 회귀 모델 생성에 필요한 기능 불러오기
from sklearn.linear_model import LinearRegression

# 아래에 코드를 채워 봅시다.
# 선형회귀 알고리즘으로 틀을 만듦
linear_model = LinearRegression()

# 훈련 데이터를 통해서 실제 인공지능 모델을 학습시킴.
linear_model.fit(X_train, y_train)

⑤ 예측 및 성능 평가

# 모델을 이용하여 예측 결과를 출력하는 코드를 작성해 봅시다.

linear_predict = linear_model.predict(X_test)

linear_predict # 저장한 예측 결과 출력하기

# 실제값과 예측값을 비교하여 결정 계수 구하기
r2_score(y_test, linear_predict)

⑥ 추가 실습: 딥러닝 회귀

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf

from sklearn.preprocessing import MinMaxScaler

# 1. 데이터 준비
X = df[['연도']].values.astype(float)
y = df[['최저시급']].values.astype(float)

# 2. 데이터 정규화
scaler_X = MinMaxScaler()
scaler_y = MinMaxScaler()

X_scaled = scaler_X.fit_transform(X)
y_scaled = scaler_y.fit_transform(y)

# 3. 딥러닝 모델 생성
model_dl = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(1,)),
    tf.keras.layers.Dense(32, activation='relu'),
    tf.keras.layers.Dense(16, activation='relu'),
    tf.keras.layers.Dense(1)
])

# 4. 모델 컴파일
model_dl.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.01),
    loss='mse'
)

# 5. 모델 학습
history = model_dl.fit(
    X_scaled,
    y_scaled,
    epochs=500,
    verbose=0
)

# 6. 그래프용 연도 생성
years = np.linspace(
    X.min(),
    2028,
    200
).reshape(-1, 1)

years_scaled = scaler_X.transform(years)

# 7. 딥러닝 예측
pred_scaled = model_dl.predict(years_scaled, verbose=0)
predictions = scaler_y.inverse_transform(pred_scaled)

# 8. 2026년과 2028년 예측
future_years = np.array([[2026], [2028]])
future_scaled = scaler_X.transform(future_years)

future_pred_scaled = model_dl.predict(future_scaled, verbose=0)
future_predictions = scaler_y.inverse_transform(future_pred_scaled)

# 9. 그래프 그리기
plt.figure(figsize=(12, 7))

# 실제 데이터
plt.scatter(
    X.flatten(),
    y.flatten(),
    color='blue',
    s=60,
    label='Actual Minimum Wage',
    zorder=3
)

# 딥러닝 예측 곡선
plt.plot(
    years.flatten(),
    predictions.flatten(),
    color='red',
    linewidth=2,
    label='Deep Learning Prediction'
)

# 미래 예측값
plt.scatter(
    future_years.flatten(),
    future_predictions.flatten(),
    color='green',
    marker='*',
    s=250,
    label='Future Predictions',
    zorder=5
)

# 예측값 표시
for year, wage in zip(
    future_years.flatten(),
    future_predictions.flatten()
):
    plt.annotate(
        f'{year}: {wage:,.0f} KRW',
        (year, wage),
        xytext=(8, 10),
        textcoords='offset points'
    )

plt.xlabel('Year')
plt.ylabel('Minimum Wage (KRW)')
plt.title('Deep Learning Minimum Wage Prediction')

plt.legend()
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

plt.plot(
