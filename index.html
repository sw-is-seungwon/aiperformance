펭귄 부리 길이 예측

펭귄 데이터의 특성을 이용해 부리 길이를 예측합니다.

STEP 01

데이터 불러오기 및 X, y 설정

# 데이터를 불러와 봅시다.
import pandas as pd
df = pd.read_csv("펭귄_데이터(처리전).csv")

df

# 펭귄의 키와 날개길이, 몸무게를 활용하여 부리길이를 예측하도록 변수를 설정해 봅시다.

# 독립 변수
X = df [['키_m', '날개길이_cm', '몸무게_kg']]

# 종속 변수
y = df['부리길이_cm']
STEP 02

정규화

# 정규화를 진행해 봅시다.

# 정규화에 필요한 기능 불러오기
from sklearn.preprocessing import MinMaxScaler

# X 데이터를 배열 형태로 변환
data_array = X.values

# 최소-최대 정규화 방법을 사용할 틀 제작
scaler = MinMaxScaler()

# X 데이터를 0~1 사이의 값으로 정규화한 값을 변수에 저장
normalized_data = scaler.fit_transform(data_array)

# 정규화한 데이터를 다시 데이터 프레임으로 변환(기존 X의 열 이름을 그대로 사용)
X = pd.DataFrame(normalized_data, columns = X.columns)

X
STEP 03

훈련·테스트 데이터 분리

# 테스트 나누기에 필요한 기능 불러오기
from sklearn.model_selection import train_test_split

# 아래에 코드를 채워 봅시다.
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.3, # 60개의 데이터가 있을 때 테스트 데이터 18개
    random_state=42
)

len(X_test)
STEP 04

모델 생성 및 학습

# 선형회귀 모델을 생성하고, 학습을 진행하는 코드를 작성해 봅시다.

# 선형 회귀 모델 생성에 필요한 기능 불러오기
from sklearn.linear_model import LinearRegression

# 아래에 코드를 채워 봅시다.
# 선형회귀 알고리즘으로 틀을 만듦
linear_model = LinearRegression()

# 훈련 데이터를 통해서 실제 인공지능 모델을 학습시킴.
linear_model.fit(X_train, y_train)
STEP 05

예측 및 성능 평가

# 모델을 이용하여 예측 결과를 출력하는 코드를 작성해 봅시다.

linear_predict = linear_model.predict(X_test)

linear_predict # 저장한 예측 결과 출력하기

# 모델의 성능을 판단하는 함수 불러오기
from sklearn.metrics import r2_score

# 실제값과 예측값을 비교하여 결정 계수 구하기
r2_score(y_test, linear_predict)
참고 · 원본 수업 코드에서는 데이터 분리 전에 전체 X를 정규화합니다. 별도 테스트 성능을 엄밀히 평가할 때는 훈련 데이터에만 정규화 기준을 학습시키는 것이 좋습니다.

STEP 05

추가 실습: 딥러닝 회귀

import numpy as np
import pandas as pd
import tensorflow as tf

from sklearn.preprocessing import MinMaxScaler

# 1. 데이터 준비
X = df[['연도']].values.astype(float)
y = df[['최저시급']].values.astype(float)

# 2. 데이터 정규화
scaler_X = MinMaxScaler()
scaler_y = MinMaxScaler()

X_scaled = scaler_X.fit_transform(X)
y_scaled = scaler_y.fit_transform(y)

# 3. 딥러닝 모델 생성
model_dl = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(1,)),
    tf.keras.layers.Dense(32, activation='relu'),
    tf.keras.layers.Dense(16, activation='relu'),
    tf.keras.layers.Dense(1)
])

# 4. 모델 컴파일
model_dl.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.01),
    loss='mse'
)

# 5. 모델 학습
history = model_dl.fit(
    X_scaled,
    y_scaled,
    epochs=500,
    verbose=0
)

# 6. 원하는 연도의 최저시급 예측
future_years = np.array([[2026], [2028]])
future_scaled = scaler_X.transform(future_years)
future_pred_scaled = model_dl.predict(future_scaled, verbose=0)
future_predictions = scaler_y.inverse_transform(future_pred_scaled)

for year, wage in zip(future_years.flatten(), future_predictions.flatten()):
    print(f'{year}년 예상 최저시급: {wage:,.0f}원')
참고 · 원본 노트북의 예측 연도(2028)와 출력 문구(2026)가 달라 해당 부분을 일치시켰습니다. 딥러닝 코드는 원본 파일에 들어 있는 추가 실습입니다. 학습에 사용한 데이터만으로 일반화 성능을 판단할 수는 없습니다.

station_df = df.groupby('지하철역')[['08시-09시 승차인원','09시-10시 승차인원']].mean() #두 열의 평균값을 지하철역끼리 묶은 것~

station_df

X, y 설정 및 모델 학습

y = station_df['09시-10시 승차인원'] # 종속변수 (영향을 받는 변수)

# ② 선형 회귀 모델 만들기
from sklearn.linear_model import LinearRegression

linear_model = LinearRegression()

# ③ 모델 학습시키기
linear_model.fit(X, y) # 회귀는 정답이 있는 지도학습! -> 정답(종속변수) 추가

print('기울기:', linear_model.coef_[0])

STEP 04

한 지하철역 예측

prediction = linear_model.predict(pd.DataFrame(
    {'08시-09시 승차인원': [passengers]}
))
print(f'{station}의 예측 승차 인원: {prediction[0]:.0f}명')

# 실제 평균 승차 인원

predicted = prediction[0]

print(f'실제 평균 승차 인원: {actual}명')
